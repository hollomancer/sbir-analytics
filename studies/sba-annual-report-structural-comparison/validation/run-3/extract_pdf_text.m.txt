#import <Foundation/Foundation.h>
#import <PDFKit/PDFKit.h>

int main(int argc, const char *argv[]) {
    @autoreleasepool {
        if (argc != 3) {
            fprintf(stderr, "usage: extract_pdf_text INPUT.pdf OUTPUT.txt\n");
            return 2;
        }
        NSString *inputPath = [NSString stringWithUTF8String:argv[1]];
        NSString *outputPath = [NSString stringWithUTF8String:argv[2]];
        PDFDocument *document = [[PDFDocument alloc] initWithURL:[NSURL fileURLWithPath:inputPath]];
        if (document == nil) {
            fprintf(stderr, "unable to open PDF\n");
            return 3;
        }
        NSMutableString *rendered = [NSMutableString string];
        for (NSUInteger index = 0; index < document.pageCount; index++) {
            [rendered appendFormat:@"=== PDF_PAGE %lu ===\n", (unsigned long)index + 1];
            NSString *text = [document pageAtIndex:index].string;
            if (text != nil) [rendered appendString:text];
            [rendered appendString:@"\n\f\n"];
        }
        NSError *error = nil;
        if (![rendered writeToFile:outputPath atomically:YES encoding:NSUTF8StringEncoding error:&error]) {
            fprintf(stderr, "%s\n", error.localizedDescription.UTF8String);
            return 4;
        }
    }
    return 0;
}
